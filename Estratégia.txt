//| Verify trend                                             |
//+------------------------------------------------------------------+
bool Strategy::Trend()
  { 
    int instart=1;
    bool output=false;
    int last=instart;
    int prior=last+1;
//---
    double inOpen=0,inClose=0;
    inOpen=iOpen(_Symbol,_Period,last);
    inClose=iClose(_Symbol,_Period,last);
//---
    double priorOpen=0,priorClose=0;
    priorOpen=iOpen(_Symbol,_Period,prior);
    priorClose=iClose(_Symbol,_Period,prior);
//---
    CandleStick *candleStick;
    candleStick=new CandleStick();
    double valuelast=0,valueprior=0;
    if(candleStick.green(inOpen,inClose)) valuelast=inOpen; else valuelast=inClose;
    if(candleStick.green(priorOpen,priorClose)) valueprior=priorOpen; else valueprior=inClose;
    if(valueprior<valuelast) output=TrendHigh(instart);
    return output;
  }

//| High trend                                                       |
//+------------------------------------------------------------------+
bool Strategy::TrendHigh(const int start)
  {
//---    
    const int quantity=100;
    const string intimenow=TimeToString(TimeCurrent());
//---
    int instart=start;
    TrendStick *supportStick[]; 
    TrendStick *resistenceStick[]; 
    ArrayResize(supportStick,quantity);
    ArrayResize(resistenceStick,quantity);
    supportStick[quantity-1]=new TrendStick();
    resistenceStick[quantity-1]=new TrendStick();
//---
    double inOpen=0,inClose=0;
    int inTrend=0;
    inOpen=iOpen(_Symbol,_Period,1);
    inClose=iClose(_Symbol,_Period,1);
//---
    bool output=false;
    CandleStick *candleStick;
    candleStick=new CandleStick();
    Thread *thread;
    thread=new Thread();
//---
    for(int value=instart; value<quantity; value++) {
//---
      inOpen=iOpen(_Symbol,_Period,value);
      inClose=iClose(_Symbol,_Period,value);
//---
      double support=0;
      if(thread.VerifyDown(quantity,intimenow,value)) {
        if(candleStick.green(inOpen,inClose)) support=inOpen; else support=inClose;
      } else {
        support=supportStick[inTrend].support;
      }
//---  
      double endurance=0;
      if(thread.VerifyUp(quantity,intimenow,value)) {
        if(candleStick.red(inOpen,inClose)) endurance=inOpen; else endurance=inClose;
      } else {
        endurance=trendStick[inTrend].endurance;
      }
//---
      if(thread.VerifySupport(quantity,intimenow,value)) {
        if(candleStick.green(inOpen,inClose)) endurance=inOpen; else endurance=inClose;
      } 
//---
      if(thread.VerifyEndurance(quantity,intimenow,value)) {
        if(candleStick.red(inOpen,inClose)) support=inOpen; else support=inClose;
      }                     
//---
      trendStick[inTrend++].Add(value,endurance,support,inTrend++);
    }
//---
    return output;
  }
