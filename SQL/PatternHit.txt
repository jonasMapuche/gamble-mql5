                                 COUNT(1) AS QUANTITY

                                     SELECT CAST (strftime('%Y%m', DATE_EVENT) AS DECIMAL) AS YEAR_MONTH,

                                                       DATE
                                                  FROM (
                                                           SELECT DISTINCT DATE,
                                                                           PATTERN
                                                             FROM Hit
                                                       )
                                                       HIT
                                            )
                                            HIT_ALL
                                 )

                           WHERE YEAR_MONTH = STRFTIME('%Y%m', DATETIME('2024-01-01', '-1 MONTHS') ) 
                           GROUP BY PATTERN
